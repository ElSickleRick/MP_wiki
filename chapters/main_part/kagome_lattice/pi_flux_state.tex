\subsubsection{Real Space}

This system is a monoclinic (\textcolor{red}{Is this correct?}) Bravais-lattice with a six site unit cell. It emerges from the bare Kagome lattice through unit cell doubling (the stranslational symmetry gets broken by the internal mean field). 
\begin{center}
	\includegraphics[width=\textwidth]{lattice-pi-flux-state.jpeg}
\end{center}

The nearest neighbour vectors remain unchanged

\begin{equation}
	\vec{\delta}_1 = 
	\begin{pmatrix}
		1 \\ 
		0
	\end{pmatrix} \quad
	\vec{\delta}_2 = \frac12
	\begin{pmatrix}
		1 \\
		\sqrt{3}
	\end{pmatrix} \quad
	\vec{\delta}_3 = \frac12 
	\begin{pmatrix}
		1 \\
		-\sqrt{3}
	\end{pmatrix}.
\end{equation}

The new primitve translations are

\begin{equation}
	\vec{t}_1 = 
	\begin{pmatrix}
		2 \\
		0
	\end{pmatrix} \quad
	\vec{t}_2 =
	\begin{pmatrix}
		-2 \\
		2\sqrt{3}
	\end{pmatrix}
\end{equation}

The real space (effective) tight-binding hamiltonian is

\begin{align}
	\begin{split}
		H = t \sum_{i\sigma}\bigg( &- a_{i\sigma}^\dagger c_{i\sigma} - c_{i\sigma}^\dagger b_{i\sigma} + b_{i\sigma}^\dagger a_{i\sigma} + a_{i\sigma}^\dagger d_{i\sigma} + d_{i\sigma}^\dagger e_{i\sigma} + e_{i\sigma}^\dagger f_{i\sigma} + f_{i\sigma}^\dagger d_{i\sigma} \\
		&+ b_{i\sigma}^\dagger c_{i+t_1;\sigma} + b_{i\sigma}^\dagger e_{i+t_1+t_2;\sigma} + f_{i\sigma}^\dagger d_{i+t_1;\sigma} + f_{i\sigma}^\dagger a_{i+t_1;\sigma} + e_{i\sigma}^\dagger c_{i-t_2;\sigma} + \text{h.c.} \bigg)
	\end{split}
\end{align}

\subsubsection{Reciprocal Space}

The reciprocal lattice vectrors are 

\begin{equation}
	\vec{b}_{1} = \pi
	\begin{pmatrix}
		1 \\
		1/\sqrt{3}
	\end{pmatrix} \quad 
	\vec{b}_2 = \pi 
	\begin{pmatrix}
		0 \\
		1/\sqrt{3}
	\end{pmatrix}
\end{equation}

After a Fourier transform the $k$-space Hamiltonian reads

\begin{equation}
	H = t \sum_{q,\sigma} \Psi_{q\sigma}^\dagger 
	\begin{pmatrix}
		0 & e^{iq_2} & -e^{-iq_3} & 0 & e^{-iq_2} & e^{iq_3} \\
		e^{-iq_2} & 0 & 2i\sin(q_1) & e^{iq_2} & 0 & 0 \\
		-e^{iq_3} & -2i\sin(q_1) & 0 & e^{-iq_3} & 0 & 0 \\
		0 & e^{-iq_2} & e^{iq_3} & 0 & e^{iq_2} & e^{-iq_3} \\
		e^{iq_2} & 0 & 0 &  e^{-iq_2} & 0 & 2\cos(q_1) \\
		e^{-iq_3} & 0 & 0 & e^{iq_3} & 2\cos(q_1) & 0 
	\end{pmatrix}
	\Psi_{q\sigma},
\end{equation}

where
\begin{equation}
	\Psi_{q\sigma}^\dagger = 
	\begin{pmatrix}
		a_{q\sigma}^\dagger & b_{q\sigma}^\dagger & c_{q\sigma}^\dagger & e_{q\sigma}^\dagger & f_{q\sigma}^\dagger & d_{q\sigma}^\dagger 
	\end{pmatrix}.
\end{equation}

\subsubsection{Linearization in reciproical space and Projection}

For now I only do the calculations for $\vec{K}$. Use  

\begin{align}
	\vec{K}\cdot\vec{\delta}_1 &= \frac{\pi}{4} \\
	\vec{K}\cdot\vec{\delta}_2 &= \frac{\pi}{4} \\
	\vec{K}\cdot\vec{\delta}_3 &= 0
\end{align}

to obtain the linearized Hamiltonian in the vicinity of $\vec{K}$:


\begin{equation}
	\begin{pmatrix}
		0 & e^{i\frac{\pi}{4}}(1+ip_2) & -1+ip_3 & 0 & e^{-i\frac{\pi}{4}}(1-ip_2) & 1+ip_3 \\
		e^{-i\frac{\pi}{4}}(1-ip_2) & 0 & \sqrt{2}i(1+p_1) & e^{i\frac{\pi}{4}}(1+ip_2) & 0 & 0 \\
		-1-ip_3 & -\sqrt{2}i(1+p_1) & 0 & 1-ip_3 & 0 & 0 \\
		0 & e^{-i\frac{\pi}{4}}(1-ip_2) & 1+ip_3 & 0 & e^{i\frac{\pi}{4}}(1+ip_2) & 1-ip_3\\
		e^{i\frac{\pi}{4}}(1+ip_2) & 0 & 0 & e^{-i\frac{\pi}{4}}(1-ip_2) & 0 & \sqrt{2}(1-p_1) \\
		1-ip_3 & 0 & 0 & 1 + ip_3 & \sqrt{2}(1-p_1) & 0
	\end{pmatrix} 
\end{equation}

\textcolor{blue}{To keep the size reasonable maybe its a good idea to split this up into zeroth and first order in p. Might also be a good choice for storytelling purposes.} \\

The shorthand means

\begin{align*}
	p_i &\equiv \vec{\delta}_i \cdot \vec{p}, 
\end{align*}

also

\begin{equation}
	e^{i\frac{\pi}{4}} = \frac{1+i}{\sqrt{2}}.
\end{equation}

The eigenvalues for the zeroth order (/directly at the $\vec{K}$-point) are 

\begin{align}
	&\varepsilon_1(\vec{K}) = -2t  &&\varepsilon_2(\vec{K}) = -2t &&\varepsilon_3(\vec{K}) = (1-\sqrt{3})t \notag\\
	&\varepsilon_4(\vec{K}) = (1-\sqrt{3})t &&\varepsilon_5(\vec{K}) = (1+\sqrt{3})t &&\varepsilon_6(\vec{K}) = (1+\sqrt{3})t
\end{align}

$\varepsilon_3$ and $\varepsilon_4$ are the relevant ones. \\

I let mathematice calculate orthonormal eigenvectors for $H$ at $\vec{K}$. The relevant ones (that is bands 3 and 4):

\begin{align}
	\vec{v}_3(\vec{K}) &= \frac{1}{2\sqrt{6-3\sqrt{3}}}
	\begin{pmatrix}
		\frac{1-\sqrt{3}}{2} + \frac{3-\sqrt{3}}{2}i, & \sqrt{2}(1-\sqrt{3}), & (-2+\sqrt{3})i, & \frac{1-\sqrt{3}}{2} + \frac{\sqrt{3}-3}{2}i, & 0, & 1  
	\end{pmatrix}^T \\
	\vec{v}_4(\vec{K}) &= \frac{1}{2\sqrt{6(7-4\sqrt{3})}}
	\begin{pmatrix}
		(3-2\sqrt{3}) +(-2+\sqrt{3})i, & 0, & i(1-\sqrt{3}), & 3-2\sqrt{3} + (2-\sqrt{3})i, & 2\sqrt{2}(2-\sqrt{3}), & 5-3\sqrt{3}
	\end{pmatrix}^T
\end{align}

The projections onto those vectors are 

\begin{align}
	\begin{split}
		&\vec{v}_3^H\cdot H_{\vec{K}} \cdot \vec{v}_3 = (1-\sqrt{3})t - \frac{2t}{\sqrt{3}}p_2 + \frac{(\sqrt{3}-1)t}{\sqrt{3}}p_3\\
		& \vec{v}_4^H\cdot H_{\vec{K}} \cdot \vec{v}_4 = (1-\sqrt{3})t +\frac{2t}{\sqrt{3}}p_2 - \frac{(\sqrt{3}-1)t}{\sqrt{3}}p_3 \\
		&\vec{v}_3^H\cdot H_{\vec{K}} \cdot \vec{v}_4 = -\frac{2t}{\sqrt{3}}p_2 - \frac{(3+\sqrt{3})t}{3}p_3 \\
		&\vec{v}_4^H\cdot H_{\vec{K}} \cdot \vec{v}_3 = -\frac{2t}{\sqrt{3}}p_2 - \frac{(3+\sqrt{3})t}{3}p_3 
	\end{split}
\end{align}

I transform these to $p_x$ and $p_y$: 

\begin{align}
	\begin{split}
		&\vec{v}_3^H\cdot H_{\vec{K}} \cdot \vec{v}_3 = (1-\sqrt{3})t + \frac{1-\sqrt{3}}{2}tp_x  - \frac{1+\sqrt{3}}{2}tp_y \\
		& \vec{v}_4^H\cdot H_{\vec{K}} \cdot \vec{v}_4 = (1-\sqrt{3})t - \frac{1-\sqrt{3}}{2}tp_x  + \frac{1+\sqrt{3}}{2}tp_y\\
		&\vec{v}_3^H\cdot H_{\vec{K}} \cdot \vec{v}_4 = -\frac{1+\sqrt{3}}{2}tp_x - \frac{1-\sqrt{3}}{2}tp_y\\
		&\vec{v}_4^H\cdot H_{\vec{K}} \cdot \vec{v}_3 = -\frac{1+\sqrt{3}}{2}tp_x - \frac{1-\sqrt{3}}{2}tp_y 
	\end{split}
\end{align}

The dispersion of this projection is 

\begin{gather}
	\begin{split}
		\varepsilon_{3\vec{K}} &= (1-\sqrt{3})t - \sqrt{2}t\sqrt{p_x^2 + p_y^2} \\
		\varepsilon_{4\vec{K}} &= (1-\sqrt{3})t + \sqrt{2}t\sqrt{p_x^2 + p_y^2}
	\end{split}
\end{gather}

Most importantly 

\begin{equation}
	v_F = \sqrt{2}t
\end{equation}

I follow the usual method defining 

\begin{align}
	\begin{split}
		\vec{w}_3 &= \alpha \vec{v}_3 + \beta \vec{v}_4 \\
		\vec{w}_4 &= \eta \vec{v}_3 + \gamma \vec{v}_4
	\end{split}
\end{align}

Orthonormality is garuanted by 

\begin{gather}
	\abs{\alpha}^2 + \abs{\beta}^2 = 1 \\
	\abs{\eta}^2 + \abs{\gamma}^2 = 1\\
	\alpha^*\eta + \beta^*\gamma = 0
\end{gather}

\textcolor{blue}{If my counting is correct, there are 8 parameters to begin with (real and imaginary parts), the equations above eliminate 1 and 1 and 2 (the last equation are really 2 equations, one for real and one for imaginary part), leaving behind 4 free parameters.} I make the following Ansatz for the leftover parameters:

\begin{align}
	\begin{split}
		\alpha &= \cos(\phi/2)e^{i(\chi+\theta/2)} e^{i\kappa} \\
		\beta &= \sin(\phi/2)e^{i(\chi+\theta/2)} \\
		\eta &= \sin(\phi/2)e^{i(\chi-\theta/2)} e^{i\kappa} \\
		\gamma &= -\cos(\phi/2)e^{i(\chi-\theta/2)}
	\end{split}
\end{align}

\textcolor{blue}{If you want to be fancy about it you can think of what kind of matrix describes the relation between the two bases and then find a suitable parametrization for it. If one thinks of these matrices in terms of groups, maybe there is a suffisticated argument to be made  why $U(1)$ appears.}. $\phi$ is a rotation angle, $\chi$ a global phase, $\theta$ the phase difference between the two states and $\kappa$ state-internal phase. It is clear that all physical quantities can not depend on $\chi$. \textcolor{blue}{At this point I might as well choose $\chi = 0$.}\\

The following relations are impotant to derive the matrix elements in the rotated basis:

\begin{align*}
	(2\abs{\alpha}^2-1) &= \cos(\phi) \\
	2\text{Re}(\alpha^*\beta) &= \sin(\phi)\cos(\kappa) \\
	\alpha^*\eta - \beta^*\gamma &= \sin(\phi)\cos(\theta) -i \sin(\phi)\sin(\theta) \\
	\alpha^*\gamma + \beta^*\eta &= \sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)  \\
	&+ i\big( \sin(\kappa)\cos(\theta) + \cos(\phi)\cos(\kappa)\sin(\theta)\big)
\end{align*}

Using these, in general (while neglecting the $\sigma_0$-term),

\begin{align}
	\vec{w}_1^H\cdot H \cdot \vec{w}_1 &\sim  \left(\vec{v}_1^H\cdot H \cdot \vec{v}_1\right)\cos(\phi)+\left(\vec{v}_1^H\cdot H \cdot \vec{v}_2\right)\sin(\phi)\cos(\kappa)
\end{align}

\begin{align}
	\begin{split}
		\vec{w}_1^H \cdot H \cdot \vec{w}_2 &=  \left(\vec{v}_1^H\cdot H \cdot \vec{v}_1\right)\sin(\phi)\cos(\theta) + \left(\vec{v}_1^H\cdot H \cdot \vec{v}_2\right)\left(\sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)\right) \\
		&+i \left[ -\left(\vec{v}_1^H\cdot H \cdot \vec{v}_1\right)\sin(\phi)\sin(\theta) + \left(\vec{v}_1^H\cdot H \cdot \vec{v}_2\right)\big( \sin(\kappa)\cos(\theta) + \cos(\phi)\cos(\kappa)\sin(\theta)\big)   \right]
	\end{split}
\end{align}

I find for the projection in the  transformed basis (I again drop the $\sigma_0$ part as it transforms trivially):

\begin{align*}
	\frac{\vec{w}_3^H \cdot H_{\vec{K}} \cdot \vec{w}_3}{\sqrt{2}t} &\sim \frac{1}{\sqrt{8}}\left((1-\sqrt{3})\cos(\phi) - (1+\sqrt{3})\sin(\phi)\cos(\kappa)\right)p_x \\
	&+ \frac{1}{\sqrt{8}}\left(-(1+\sqrt{3})\cos(\phi) - (1-\sqrt{3})\sin(\phi)\cos(\kappa)\right)p_y
\end{align*}

\begin{align*}
	\frac{\vec{w}_4^H \cdot H_{\vec{K}} \cdot \vec{w}_4}{\sqrt{2}t} &\sim \frac{1}{\sqrt{8}} \left(-(1-\sqrt{3})\cos(\phi) +(1+\sqrt{3})\sin(\phi)\cos(\kappa)\right)p_x \\
	&+ \frac{1}{\sqrt{8}} \left((1+\sqrt{3})\cos(\phi) + (1-\sqrt{3})\sin(\phi)\cos(\kappa)\right) p_y
\end{align*}

\begin{align*}
	\frac{\vec{w}_3^H \cdot H_{\vec{K}} \cdot \vec{w}_4}{\sqrt{2}t} &= \frac{1}{\sqrt{8}}\left((1-\sqrt{3})\sin(\phi)\cos(\theta) - (1+\sqrt{3})\left[\sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)\right]\right)p_x \\
	&+\frac{1}{\sqrt{8}}\left(-(1+\sqrt{3})\sin(\phi)\cos(\theta) - (1-\sqrt{3})\left[\sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)\right]\right) p_y \\
	&-\frac{i}{\sqrt{8}}\left((1-\sqrt{3})\sin(\phi)\sin(\theta) + (1+\sqrt{3})\left[\sin(\kappa)\cos(\theta)+\cos(\phi)\cos(\kappa)\sin(\theta)\right]\right) p_x \\
	&-\frac{i}{\sqrt{8}} \left(-(1+\sqrt{3})\sin(\phi)\sin(\theta) + (1-\sqrt{3})\left[\sin(\kappa)\cos(\theta)+\cos(\phi)\cos(\kappa)\sin(\theta)\right]\right) p_y 
\end{align*}


\begin{align*}
	\frac{\vec{w}_4^H \cdot H_{\vec{K}}\cdot \vec{w}_3}{\sqrt{2}t}  &= \frac{1}{\sqrt{8}}\left((1-\sqrt{3})\sin(\phi)\cos(\theta) - (1+\sqrt{3})\left[\sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)\right]\right)p_x \\
	&+\frac{1}{\sqrt{8}}\left(-(1+\sqrt{3})\sin(\phi)\cos(\theta) - (1-\sqrt{3})\left[ \sin(\kappa)\sin(\theta)-\cos(\phi)\cos(\kappa)\cos(\theta)\right]\right)p_y \\
	&+\frac{i}{\sqrt{8}}\left( (1-\sqrt{3})\sin(\phi)\sin(\theta) + (1+\sqrt{3})\left[\sin(\kappa)\cos(\theta) + \cos(\phi)\cos(\kappa)\sin(\theta)\right]\right)p_x \\
	&+\frac{i}{\sqrt{8}}\left(-(1+\sqrt{3})\sin(\phi)\sin(\theta) + (1-\sqrt{3})\left[\sin(\kappa)\cos(\theta) + \cos(\phi)\cos(\kappa)\sin(\theta)\right]\right)p_y
\end{align*}

From these equations the $\sigma_{0,x,y,z}$-components can be read of. I decided to solve for an equation of the form $H=\varepsilon_0\sigma_0 + v_F(p_x \sigma_x + p_y \sigma_z)$. The corresponding angles are  \\

\begin{align}
	\begin{split}
		\cos(\phi) &= -\frac{1+\sqrt{3}}{\sqrt{8}} \quad \Rightarrow \quad \phi = \frac{11}{12}\pi\\
		\cos(\kappa) &= - \frac{1-\sqrt{3}}{\sqrt{4-2\sqrt{3}}} = 1 \quad \Rightarrow \quad \kappa = 0\\
		\cos(\theta) &= \frac{1-\sqrt{3}}{\sqrt{4-2\sqrt{3}}} = -1\quad \Rightarrow \quad \theta = \pi.
	\end{split}
\end{align}


\textcolor{blue}{$\sqrt{4-2\sqrt{3}} = \sqrt{3}-1$ :D. First I had lhs=$1-\sqrt{3}$, and it took me around 3 hours to find the mistake.} \\


Another solution corresponding to the form $H = \varepsilon_0\sigma_0 + v_F(p_x\sigma_x +p_y\sigma_y)$ is

\begin{align}
	\begin{split}
		\phi = -\frac{\pi}{2} \\
		\kappa = -\frac{\pi}{2} \\
		\theta = \frac{5}{12}\pi
	\end{split}
\end{align}

\subsubsection{Introduction of strain} 

The relevant equations are 

\begin{equation}
	t(\vec{\delta}_{ij})e^{i\vec{K}\cdot\vec{\delta}_{ij}}  \approx t e^{i\vec{K}\cdot\vec{\delta}_{ij}}\left(1+i\vec{p}\cdot\vec{\delta}_{ij} - \beta \left( \vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right) + i \left(\vec{K} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right)\right)
\end{equation}

and 

\begin{align}
	\begin{split}
		t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}} + t(-\vec{\delta}_{ij})e^{-i\vec{K}\cdot\vec{\delta}_{ij}}  \approx & 2t\cos(	\vec{q}\cdot\vec{\delta}_{ij}) - 2t\sin(\vec{q} \cdot \vec{\delta}_{ij})\left(\vec{p}\cdot\vec{\delta}_{ij}\right) \\
		-&2\beta t\cos(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij} \right) \\
		-&2t\sin(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{K} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right)
	\end{split}
\end{align}

and 

\begin{align}
	\begin{split}
		t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}}-t(-\vec{\delta}_{ij})e^{-i\vec{q}\cdot\vec{\delta}_{ij}} \approx & 2it \sin(\vec{K}\cdot\vec{\delta}_{ij}) +2it\cos(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{p}\cdot\vec{\delta}_{ij}\right) \\
		-&2i\beta t \sin(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right) \\
		+&2it\cos(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{K}\cdot \underline{u} \cdot \vec{\delta}_{ij}\right)
	\end{split}
\end{align}

I split the everything into two parts, the first one being proportional to $\beta$:

\begin{equation}
	\langle \delta H_1 \rangle_{\vec{w}_1, \vec{w}_2} = -\frac{\beta}{2}(1-\sqrt{3})(u_{xx}+u_{yy})\sigma_0.
\end{equation}

\textcolor{red}{There is also a $\beta^0$-term that I shouldwrite down. However, this term can be gauged away so it not relevant for the physical discussion.} \\

This means that to this order there is no effective vector potential but only an effective scalar potential.
