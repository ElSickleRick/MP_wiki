\subsubsection{Real Space}

The bare Kagome lattice is a triangular Bravais lattice with a three site unit cell (called a, b and c-sites below). The Wigner Seitz-cell is a hexagon (with full symmetries, even if the drawing suggests otherwise ;) ). Geometrically speaking the lattice consists of corner-sharing equilateral triangles.


\begin{center}
	\includegraphics[width = \textwidth]{lattice-0-flux-state.jpeg}
\end{center}

In all that follows I normalize all distances wrt the distance between nearest neighbours. The nearest neighbour vectors are ( see cartoon)

\begin{equation}
	\vec{\delta}_1 = 
	\begin{pmatrix}
		1 \\
		0
	\end{pmatrix} \quad 
	\vec{\delta}_2 = \frac12
	\begin{pmatrix}
		1 \\
		\sqrt{3}
	\end{pmatrix} \quad
	\vec{\delta}_3 = \frac12 
	\begin{pmatrix}
		1 \\
		-\sqrt{3}
	\end{pmatrix}
\end{equation}

and the primitive translations 

\begin{equation}
	\vec{t}_1 = 2\vec{\delta}_1 =
	\begin{pmatrix}
		2 \\
		0
	\end{pmatrix} \quad 
	\vec{t}_2 = 2\vec{\delta}_2 =
	\begin{pmatrix}
		1 \\
		\sqrt{3}
	\end{pmatrix}.
\end{equation}

The tight binding hamiltonain is 

\begin{equation}
	H = t \sum_{i\sigma} \left(a_{i\sigma}^\dagger c_{i\sigma} + c_{i\sigma}^\dagger b_{i\sigma} + b_{i\sigma}^\dagger a_{i\sigma} + c_{i\sigma}^\dagger b_{i+t_1;\sigma} + c_{i\sigma}^\dagger a_{i+t_1-t_2;\sigma} + a_{i\sigma}^\dagger b_{i+t_2;\sigma} + \text{h.c.}\right)
\end{equation}

The first index refers to the unit cell and the second one to the "species" of the particle, the latter being a product of the parton decomposition.

\subsubsection{Reciprocal Space}

The recipricoal lattice vectors are 


\begin{equation}
	\vec{b}_1 = \pi
	\begin{pmatrix}
		1 \\
		-1/\sqrt{3}
	\end{pmatrix}, \quad
	\vec{b}_2 = \pi 
	\begin{pmatrix}
		0 \\
		2/\sqrt{3}
	\end{pmatrix}
\end{equation}

The first Brillouin zone is also a hexagon (rotated by 90° wrt to the Wigner-Seitz cell). To find $H$ in reciprocal space I do a Fourier transform: 

\begin{equation}
	a_{i\sigma} = \sqrt{\frac{3}{N}} \sum_{k} e^{i\vec{k}\vec{R}_i} a_{k\sigma},
\end{equation}

where $N$ is the number of unit cells. This yields

\begin{equation}
	H = 2t \sum_{q\sigma}
	\begin{pmatrix}
		a_{q\sigma}^\dagger & b_{q\sigma}^\dagger & c_{q\sigma}^\dagger 
	\end{pmatrix}
	\begin{pmatrix}
		0 & \cos(\vec{q}\cdot\vec{\delta}_2) & \cos(\vec{q}\cdot\vec{\delta}_3) \\
		\cos(\vec{q}\cdot\vec{\delta}_2) & 0 & \cos(\vec{q}\cdot\vec{\delta}_1) \\
		\cos(\vec{q}\cdot\vec{\delta}_3) & \cos(\vec{q}\cdot\vec{\delta}_1) & 0
	\end{pmatrix}
	\begin{pmatrix}
		a_{q\sigma} \\
		b_{q\sigma} \\
		c_{q\sigma}
	\end{pmatrix}.
\end{equation}

\subsubsection{dispersion and band touching points}

The three eigenvalues are 

\begin{align}
	\varepsilon_1(\vec{q}) &= t\left(1-\sqrt{4\cos^2(q_x)+4\cos(q_x) \cos(\sqrt{3}q_y)+1}\right) \\
	\varepsilon_2(\vec{q}) &= t\left(1+\sqrt{4\cos^2(q_x)+4\cos(q_x) \cos(\sqrt{3}q_y)+1}\right) \\
	\varepsilon_3(\vec{q}) &= -2t,
\end{align}

see \cite{DiSante2026}. I also used 

\begin{equation}
	\cos^2\left(\frac12\left( q_x + \sqrt{3}q_y\right)\right) + \cos^2\left(\frac12\left(q_x - \sqrt{3}q_y\right)\right) = \cos(q_x)\cos(\sqrt{3}q_y) + 1
\end{equation}.

Interestingly, the top band is completely flat. The other two bands touch when

\begin{equation}
	\cos^2(q_x) + \cos(q_x)\cos(\sqrt{3}q_y) + 1/4 = 0
\end{equation}

\textcolor{gray}{
	\begin{gather}
		\Rightarrow \cos(q_x) = \frac12\left(-\cos(\sqrt{3}q_y) \pm \sqrt{\cos^2(\sqrt{3}q_y)-1}\right) \\
		\Rightarrow q_y = \frac{\pi l}{\sqrt{3}}, \; l \in \mathbb{N} \\
		\Rightarrow q_x = (-1)^s\left(\frac{\pi}{2} + \frac{(-1)^{l}\pi}{6}\right) + 2\pi m , \; m \in \mathbb{N}, \; s \in [0,1]
	\end{gather}
}

A handy solution to this equation is 

\begin{align}
	\begin{split}
		q_x &= (-1)^s\frac{2\pi}{3} + m_1\pi \\
		q_y &= \frac{\pi}{\sqrt{3}}\left(2m_2-m_1\right),
	\end{split}
\end{align}

with $s \in {0,1}$ and $m_1, m_2 \in \mathbb{N}$. These touching points lie at the corners of the Brillouin zone, they come in two flavours. \textcolor{blue}{The parameter s labels the afformentioned flavours while $m_1$ ($m_2$) is a translation by the reciprocal lattice vector $\vec{b}_1$ ($\vec{b}_2$).}  For example 

\begin{align}
	\vec{K} &= 
	\begin{pmatrix}
		2\pi/3 \\
		0
	\end{pmatrix}, \quad s=0, \; m_1 = 0, \; m_2 = 0 \\
	\vec{K}^\prime &= 
	\begin{pmatrix}
		-2\pi/3 \\
		0
	\end{pmatrix}, \quad s = 1, \; m_1 = 0, \; m_2 = 0.
\end{align}

\subsubsection{Linearization in Reciprocal Space and Projection}

Use

\begin{equation}
	\cos(\left(\vec{K}^{(\prime)}+\vec{p}\right)\cdot\vec{\delta}_{ij}) \approx \cos(\vec{K}^{(\prime)}\cdot\vec{\delta}_{ij}) - \sin(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{p}\cdot\vec{\delta}_{ij}\right)
\end{equation}


and

\begin{align}
	\begin{split}
		\vec{K}\cdot\vec{\delta}_{1} &= \frac{2\pi}{3} \\
		\vec{K}\cdot\vec{\delta}_{2} &= \frac{\pi}{3} \\
		\vec{K}\cdot\vec{\delta}_{3} &= \frac{\pi}{3} \\
		\vec{K}^\prime &= - \vec{K}
	\end{split}
\end{align}


\begin{align}
	H_{\vec{K}} \approx t\sum_{p\sigma}
	\begin{pmatrix}
		a_{p\sigma}^\dagger & b_{p\sigma}^\dagger & c_{p\sigma}^\dagger
	\end{pmatrix}
	\begin{pmatrix}
		0 & 1-\sqrt{3}\vec{p}\cdot\vec{\delta}_2 & 1 - \sqrt{3}\vec{p}\cdot\vec{\delta}_3 \\
		1-\sqrt{3}\vec{p}\cdot\vec{\delta}_2 & 0 & -1-\sqrt{3}\vec{p}\cdot\vec{\delta}_1 \\
		1-\sqrt{3}\vec{p}\cdot\vec{\delta}_3 & -1-\sqrt{3}\vec{p}\cdot\vec{\delta}_1 & 0
	\end{pmatrix}
	\begin{pmatrix}
		a_{q\sigma} \\
		b_{q\sigma} \\
		c_{q\sigma}
	\end{pmatrix}
\end{align}

Directly at the $\vec{K}$-point the eigenvalues are obviously 

\begin{equation}
	\varepsilon_1(\vec{K}) = t, \quad \varepsilon_2(\vec{K}) = t, \quad \varepsilon_3(\vec{K}) = -2t,
\end{equation}

the eigenvector corresponding to the flat band reads at the $\vec{K}$-point:

\begin{equation}
	\vec{v}_3(\vec{K}) = \frac{1}{\sqrt{3}}
	\begin{pmatrix}
		-1 \\
		1 \\
		1
	\end{pmatrix}.
\end{equation}

Since $H$ is hermitian (/symmetric in this case) there exsit a orthonormal basis of eigenvectors. Obviously the subspace orthonormal to $\vec{v}_3(\vec{K})$ corresponds to the two bands that cross. A possible (orthonormal) basis for this subspace is 

\begin{equation}
	\vec{v}_1(\vec{K}) = \frac{1}{\sqrt{2}}
	\begin{pmatrix}
		1 \\
		1 \\ 
		0
	\end{pmatrix}, \quad
	\vec{v}_2(\vec{K}) = \frac{1}{\sqrt{6}}
	\begin{pmatrix}
		1 \\
		-1 \\
		2
	\end{pmatrix}.
\end{equation}

Projeting $H$ onto this basis yields

\begin{align*}
		& \vec{v}_1^T \cdot H_{\vec{K}} \cdot \vec{v}_1 = t-\sqrt{3}t\vec{\delta}_2 \cdot \vec{p} & \vec{v}_1^T \cdot H_{\vec{K}} \cdot \vec{v}_2 = -t\left(\vec{\delta}_1 + \vec{\delta}_3\right)\cdot\vec{p} \\ 
		& \vec{v}_2^T \cdot H_{\vec{K}} \cdot \vec{v}_1 = -t\left(\vec{\delta}_1 + \vec{\delta}_3\right)\cdot\vec{p} & \vec{v}^T_2 \cdot H_{\vec{K}} \cdot \vec{v}_2 = t+\sqrt{3}t\vec{\delta}_2\cdot\vec{p}
\end{align*}

\begin{equation}
	\Leftrightarrow t\sigma_0 - \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\sigma_z + t\left(\left(\vec{\delta}_1 + \vec{\delta}_3\right)\cdot\vec{p}\right) \sigma_x
\end{equation}

This already has the correct dispersiob

\begin{equation}
	\varepsilon_{1,2}(\vec{K}+\vec{p}) \approx \sqrt{3}t\abs{\vec{p}},
\end{equation}

but is not yet in the form of the Weyl equation. To transform the equation into the Weyl equation one can rotate the basis inside the degenerate subspace. I take the following approach; define 

\begin{align}
	\begin{split}
		\vec{w}_1 &= \alpha \vec{v}_1 + \beta\vec{v}_2 \\
		\vec{w}_2 &= \eta \vec{v}_1 + \gamma \vec{v}_2
	\end{split}
\end{align}

Othogonality dictates

\begin{align*}
	\abs{\alpha}^2 + \abs{\beta}^2 = 1 \\
	\abs{\eta}^2 + \abs{\gamma}^2 = 1 \\
\alpha^* \eta + \beta^* \gamma = 0.
\end{align*}

\textcolor{blue}{Maybe make an Ansatz using trig functions and phases? Though I should count free parameters before. Educated guess with 3 free parameters:
\begin{align*}
	\alpha &= e^{i\theta_1}\sin(\phi) \\
	\beta &= e^{i\theta_2}\cos(\phi) \\
	\eta &= e^{i\theta_1}\cos(\phi) \\
	\gamma &= -e^{i\theta_2}\sin(\phi).
\end{align*}} \\

Utilizing $\vec{\delta}_2 + \vec{\delta}_3 = \vec{\delta}_1$ I get:

\begin{align}
	\vec{w}_1^H \cdot H_{\vec{K}} \cdot \vec{w}_1 &= t + \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(1-2\abs{\alpha}^2-\frac{2\text{Re}(\alpha^*\beta)}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left( -\frac{4\text{Re}(\alpha^*\beta)}{\sqrt{3}}\right) \\
	\vec{w}_2^H \cdot H_{\vec{K}} \cdot \vec{w}_2 &= 1 + \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(1-2\abs{\eta}^2-\frac{2\text{Re}(\eta^*\gamma)}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left(-\frac{4\text{Re}(\eta^*\gamma)}{\sqrt{3}}\right) \\
	\vec{w}_1^H \cdot H_{\vec{K}} \cdot \vec{w}_2 &= \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(-2\alpha^*\eta-\frac{\alpha^*\gamma + \beta^*\eta}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left(-\frac{2(\alpha^*\gamma +  \beta^*\eta)}{\sqrt{3}}\right) \\
	\vec{w}_2^H \cdot H_{\vec{K}} \cdot \vec{w}_1 &= \left(\vec{w}_1^H \cdot H_{\vec{K}} \cdot \vec{w}_2\right)^*
\end{align}

The strategy is to solve for values for $\alpha, \beta, \gamma$ and $\eta$ such that the following combinations occur in the matrix elements:

\begin{align}
	p_x &= \left( \vec{\delta}_2 + \vec{\delta}_3 \right) \cdot \vec{p} \\
	p_y &= \frac{1}{\sqrt{3}} \left(\vec{\delta}_2 - \vec{\delta}_3\right) \cdot \vec{p}
\end{align}

whil still leading to orthonormal basis vectors. The following parameters are a sensible choice 

\begin{equation}
	\alpha = -\frac12, \quad \beta = \frac{\sqrt{3}}{2}, \quad \eta = \frac{\sqrt{3}}{2}, \quad \gamma = \frac12.
\end{equation}

Thus, the new basis vectors read

\begin{equation}
	\vec{w}_1 = \frac{1}{\sqrt{2}}
	\begin{pmatrix}
		0 \\ -1 \\ 1
	\end{pmatrix}, \quad
	\vec{w}_2 = \frac{1}{\sqrt{6}}
	\begin{pmatrix}
		2 \\ 1 \\ 1
	\end{pmatrix}
\end{equation}

\begin{equation}
	\Rightarrow \langle H_{\vec{K}} \rangle_{\vec{w}_1, \vec{w}_2} = t\sigma_0 + \sqrt{3}t p_y \sigma_x + \sqrt{3}tp_x \sigma_z
\end{equation}

This form was also found in \cite{Guo2009}, a different one was obtained in \cite{Liu2020}. \textcolor{blue}{I expect that I would find the latter if I chose different values for $\alpha$, $\beta$, $\gamma$ and $\eta$.} I decided to stick with this form to keep the expressions real. \\

\subsubsection{Introduction of Strain}

I show how to expand expressions prototyical for the k-space Hamiltonian. First I expand in strain:

\begin{equation}
	t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}}  \approx te^{i\vec{q}\cdot\vec{\delta}_{ij}}\left(1-\beta\left(\vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij} \right) + i \left(\vec{q}\cdot\underline{u}\cdot\vec{\delta}_{ij}\right)\right),
\end{equation}

afterwars I expand around $\vec{K}$:

\begin{equation}
	t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}}  \approx t e^{i\vec{q}\cdot\vec{\delta}_{ij}}\left(1 + \vec{p} \cdot \vec{\delta}_{ij} \right)\left(1-\beta\left(\vec{\delta}_{ij}\cdot\underline{u}\cdot\vec{\delta}_{ij}\right) + i \left(\vec{K}\cdot\underline{u}\cdot\vec{\delta}_{ij}\right) + i \left(\underline{u}\cdot\vec{\delta}_{ij}\right)\cdot\vec{p} \right).
\end{equation}

\textcolor{blue}{This compares well with the field theory method for the triangular lattice where I did the expansions in reverse order.} \\


Keeping only the relevant terms (relevant to this order at least):

\begin{equation}
	t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}}  \approx t e^{i\vec{K}\cdot\vec{\delta}_{ij}}\left(1+i\vec{p}\cdot\vec{\delta}_{ij} - \beta \left( \vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right) + i \left(\vec{K} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right)\right)
\end{equation}

\begin{align}\label{strain_elements}
	\begin{split}
	\Rightarrow t(-\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}} + t(\vec{\delta}_{ij})e^{i\vec{q}\cdot\vec{\delta}_{ij}} \approx & 2t\cos(\vec{K}\cdot\vec{\delta}_{ij}) - 2t\sin(\vec{K} \cdot \vec{\delta}_{ij})\left(\vec{p}\cdot\vec{\delta}_{ij}\right) \\
		-&2\beta t\cos(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{\delta}_{ij} \cdot \underline{u} \cdot \vec{\delta}_{ij} \right) \\
		-&2t\sin(\vec{K}\cdot\vec{\delta}_{ij})\left(\vec{K} \cdot \underline{u} \cdot \vec{\delta}_{ij}\right)
	\end{split}
\end{align}

I group the linear corrections by order in $\beta$ to find the two matrices:

\begin{equation}
	\delta H_1 = -2t\beta
	\begin{pmatrix}
		0 & (u_{xx} + 2\sqrt{3}u_{yx}+3u_{yy})/8 & (u_{xx} - 2\sqrt{3}u_{yx}+3u_{yy}/8 \\
		(u_{xx} + 2\sqrt{3}u_{yx}+3u_{yy})/8 & 0 & -u_{xx}/2 \\
		(u_{xx} - 2\sqrt{3}u_{yx}+3u_{yy}/8 & - u_{xx}/2 & 0
	\end{pmatrix}
\end{equation}

\begin{equation}
	\delta H_2 = -2t
	\begin{pmatrix}
		0 & \pi(\frac{u_{xx}}{\sqrt{3}} + u_{yx})/2 & \pi(\frac{u_{xx}}{\sqrt{3}} - u_{yx} )/2 \\
		\pi(\frac{u_{xx}}{\sqrt{3}} + u_{yx})/2 & 0 & \pi \frac{u_{xx}}{\sqrt{3}} \\
		\pi(\frac{u_{xx}}{\sqrt{3}} - u_{yx} )/2 & \pi \frac{u_{xx}}{\sqrt{3}} & 0 
	\end{pmatrix}
\end{equation}

The projections are 

\begin{equation}
	\langle \delta H_1 \rangle_{\vec{w}_1, \vec{w}_2} = -t\beta
	\begin{pmatrix}
		u_{xx} & u_{yx} \\
		u_{yx} & u_{yy}
	\end{pmatrix}
\end{equation}

\begin{equation}
	\langle \delta H_2 \rangle_{\vec{w}_1, \vec{w}_2} = -\frac{2\pi t}{\sqrt{3}}
	\begin{pmatrix}
		u_{xx} & u_{yx} \\
		u_{yx} & -u_{xx}
	\end{pmatrix}
	= -\frac{2\pi t}{\sqrt{3}} \left( u_{yx} \sigma_y + u_{xx} \sigma_z\right)
\end{equation}

$\langle \delta H_1 \rangle_{\vec{w}_1, \vec{w}_2}$ can be written as vector potential + scalar potential using:

\begin{align}
	\begin{split}
		u_{xx} &= \frac12\left(u_{xx} - u_{yy}\right) + \frac12\left(u_{xx} + u_{yy} \right) \\
		u_{yy} &= -\frac12\left(u_{xx} - u_{yy}\right) + \frac12\left(u_{xx} + u_{yy} \right). 	
	\end{split}
\end{align}

\begin{equation}
	\Rightarrow \langle H_{\vec{K}} + \delta H_1  + \delta H_2 \rangle_{\vec{w}_1, \vec{w}_2} = (t+\phi)\sigma^0 + \sqrt{3}t\left(p_y + A_{\vec{K}}^y\right)\sigma_x + \sqrt{3}t \left( p_x + A_{\vec{K}}^x\right)\sigma_z
\end{equation}


\begin{align}
	\phi &= \frac{t}{2}\left(u_{xx}+u_{yy}\right) \\
	A_{\vec{K}}^x &= -\frac{t\beta}{2\sqrt{3}}\left(u_{xx}-u_{yy}\right) - \frac{2\pi t}{3}u_{xx} \\
	A_{\vec{K}}^y &= -\frac{t\beta}{\sqrt{3}}u_{yx} - \frac{2\pi t}{3}u_{yx}
\end{align}

The $\beta$-independent part of $\vec{A}_{\vec{K}}$ can be eliminated using a gauge-transforamtion. This agrees with the results here \cite{Liu2020}. \textcolor{red}{It bothers me that only $u_{yx}$ appears but not $u_{xy}$. Check this}\\

I give a few intermediate results for the derviation of $\vec{A}_{\vec{K}^\prime}$: \\

\textcolor{blue}{The cool kids introduce a cone flavour index and derive the expressioons for both flavours at the same time}

\begin{align}
	H_{\vec{K}^\prime} \approx t\sum_{p\sigma}
	\begin{pmatrix}
		a_{p\sigma}^\dagger & b_{p\sigma}^\dagger & c_{p\sigma}^\dagger
	\end{pmatrix}
	\begin{pmatrix}
		0 & 1+\sqrt{3}\vec{p}\cdot\vec{\delta}_2 & 1 + \sqrt{3}\vec{p}\cdot\vec{\delta}_3 \\
		1+\sqrt{3}\vec{p}\cdot\vec{\delta}_2 & 0 & -1+\sqrt{3}\vec{p}\cdot\vec{\delta}_1 \\
		1+\sqrt{3}\vec{p}\cdot\vec{\delta}_3 & -1+\sqrt{3}\vec{p}\cdot\vec{\delta}_1 & 0
	\end{pmatrix}
	\begin{pmatrix}
		a_{q\sigma} \\
		b_{q\sigma} \\
		c_{q\sigma}
	\end{pmatrix}
\end{align}

(the zeroth order is the same so I make the same Ansatz for the initial basis.)



\begin{align*}
		& \vec{v}_1^T \cdot H_{\vec{K}^\prime} \cdot \vec{v}_1 = t+\sqrt{3}t\vec{\delta}_2 \cdot \vec{p} & \vec{v}_1^T \cdot H_{\vec{K}^\prime} \cdot \vec{v}_2 = t\left(\vec{\delta}_1 + \vec{\delta}_3\right)\cdot\vec{p} \\ 
		& \vec{v}_2^T \cdot H_{\vec{K}^\prime} \cdot \vec{v}_1 = t\left(\vec{\delta}_1 + \vec{\delta}_3\right)\cdot\vec{p} & \vec{v}^T_2 \cdot H_{\vec{K}^\prime} \cdot \vec{v}_2 = t-\sqrt{3}t\vec{\delta}_2\cdot\vec{p}
\end{align*}


\begin{align}
	\vec{w}_1^H \cdot H_{\vec{K}^\prime} \cdot \vec{w}_1 &= t + \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(-1+2\abs{\alpha}^2+\frac{2\text{Re}(\alpha^*\beta)}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left( \frac{4\text{Re}(\alpha^*\beta)}{\sqrt{3}}\right) \\
	\vec{w}_2^H \cdot H_{\vec{K}^\prime} \cdot \vec{w}_2 &= 1 + \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(-1+2\abs{\eta}^2+\frac{2\text{Re}(\eta^*\gamma)}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left(\frac{4\text{Re}(\eta^*\gamma)}{\sqrt{3}}\right) \\
	\vec{w}_1^H \cdot H_{\vec{K}^\prime} \cdot \vec{w}_2 &= \sqrt{3}t\left(\vec{\delta}_2\cdot\vec{p}\right)\left(2\alpha^*\eta+\frac{\alpha^*\gamma + \beta^*\eta}{\sqrt{3}}\right) + \sqrt{3}t\left(\vec{\delta}_3\cdot\vec{p}\right)\left(\frac{2(\alpha^*\gamma +  \beta^*\eta)}{\sqrt{3}}\right) \\

\begin{equation}

\end{equation}

\begin{equation}
	\Rightarrow \vec{w}_1 = \frac{1}{\sqrt{6}}
	\begin{pmatrix}

		1 \\
		1
	\end{pmatrix}, \quad 
	\vec{w}_2 = \frac{1}{\sqrt{2}}
	\begin{pmatrix}
		0 \\
		1 \\ 
		-1
	\end{pmatrix}
\end{equation}

These are the same basis vectors as for $\vec{K}$ just in reverse order.

\begin{equation}
	\Rightarrow \langle H_{\vec{K}^\prime} \rangle_{\vec{w}_1, \vec{w}_2} = t\sigma_0 + \sqrt{3}tp_y\sigma_x + \sqrt{3}tp_x\sigma_z
\end{equation}

$\delta H_{1}$ and $\delta H_{2}$ are the same as for $\vec{K}^\prime$ (see \ref{strain_elements} and subsitiute $\vec{K} \rightarrow \vec{K}^\prime = \vec{K}$). With the reversed order of the basis vectors this gives

	\langle \delta H_1 \rangle_{\vec{w}_1, \vec{w}_2} = -t\beta

	\end{pmatrix}
\end{equation}

\begin{equation}
	\langle \delta H_2 \rangle_{\vec{w}_1, \vec{w}_2} = -\frac{2\pi t}{\sqrt{3}}
	\begin{pmatrix}
		-u_{xx} & u_{yx} \\
		u_{yx} & u_{xx}
	\end{pmatrix}
	= -\frac{2\pi t}{\sqrt{3}} \left( u_{yx} \sigma_y - u_{xx} \sigma_z\right)
