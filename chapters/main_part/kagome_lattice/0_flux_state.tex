\subsubsection{Real Space}

The bare Kagome lattice is a triangular Bravais lattice with a three site unit cell (called a, b and c-sites below). The Wigner Seitz-cell is a hexagon (with full symmetries, even if the drawing suggests otherwise :) ). Geometrically speaking the lattice consists of corner-sharing equilateral triangles.


\begin{center}
	\includegraphics[width = \textwidth]{lattice-0-flux-state.jpeg}
\end{center}

In all that follows I normalize all distances wrt the distance between nearest neighbours. The nearest neighbour vectors are ( see cartoon)

\begin{equation}
	\vec{\delta}_1 = 
	\begin{pmatrix}
		1 \\
		0
	\end{pmatrix} \quad 
	\vec{\delta}_2 = \frac12
	\begin{pmatrix}
		1 \\
		\sqrt{3}
	\end{pmatrix} \quad
	\vec{\delta}_3 = \frac12 
	\begin{pmatrix}
		1 \\
		-\sqrt{3}
	\end{pmatrix}
\end{equation}

and the primitive translations 

\begin{equation}
	\vec{t}_1 = 2\vec{\delta}_1 =
	\begin{pmatrix}
		2 \\
		0
	\end{pmatrix} \quad 
	\vec{t}_2 = 2\vec{\delta}_2 =
	\begin{pmatrix}
		1 \\
		\sqrt{3}
	\end{pmatrix}.
\end{equation}

The tight binding hamiltonain is 

\begin{equation}
	H = t \sum_{i\sigma} \left(a_{i\sigma}^\dagger c_{i\sigma} + c_{i\sigma}^\dagger b_{i\sigma} + b_{i\sigma}^\dagger a_{i\sigma} + c_{i\sigma}^\dagger b_{i+t_1;\sigma} + c_{i\sigma}^\dagger a_{i+t_1-t_2;\sigma} + a_{i\sigma}^\dagger b_{i+t_2;\sigma} + \text{h.c.}\right)
\end{equation}

The first index refers to the unit cell and the second one to the "species" of the particle, the latter being a product of the parton decomposition.

\subsubsection{Reciprocal Space and Bands}

The recipricoal lattice vectors are 

\begin{equation}
	\vec{b}_1 = \pi
	\begin{pmatrix}
		1 \\
		-1/\sqrt{3}
	\end{pmatrix}, \quad
	\vec{b}_2 = \pi 
	\begin{pmatrix}
		0 \\
		2/\sqrt{3}
	\end{pmatrix}
\end{equation}

The first Brillouin zone is also a hexagon (rotated by 90° wrt to the Wigner-Seitz cell). To find $H$ in reciprocal space I do a Fourier transform: 

\begin{equation}
	a_{i\sigma} = \frac{1}{\sqrt{N}} \sum_{k} e^{i\vec{k}\vec{R}_i} a_{k\sigma},
\end{equation}

where $N$ is the number of unit cells. This yields

\begin{equation}
	H = 2t \sum_{q\sigma}
	\begin{pmatrix}
		a_{q\sigma}^\dagger & b_{q\sigma}^\dagger & c_{q\sigma}^\dagger 
	\end{pmatrix}
	\begin{pmatrix}
		0 & \cos(\vec{q}\cdot\vec{\delta}_2) & \cos(\vec{q}\cdot\vec{\delta}_3) \\
		\cos(\vec{q}\cdot\vec{\delta}_2) & 0 & \cos(\vec{q}\cdot\vec{\delta}_1) \\
		\cos(\vec{q}\cdot\vec{\delta}_3) & \cos(\vec{q}\cdot\vec{\delta}_1) & 0
	\end{pmatrix}
	\begin{pmatrix}
		a_{q\sigma} \\
		b_{q\sigma} \\
		c_{q\sigma}
	\end{pmatrix}.
\end{equation}

\paragraph{Bands} \hfill \\

The three bands are 

\begin{align}
	\varepsilon_1(\vec{q}) &= t\left(1-\sqrt{4\cos^2(q_x)+4\cos(q_x) \cos(\sqrt{3}q_y)+1}\right) \\
	\varepsilon_2(\vec{q}) &= t\left(1+\sqrt{4\cos^2(q_x)+4\cos(q_x) \cos(\sqrt{3}q_y)+1}\right) \\
	\varepsilon_3(\vec{q}) &= -2t,
\end{align}

see \cite{DiSante2026}. Interestingly, the top band is completely flat. 

\paragraph{Dirac points} \hfill \\

The other two bands touch when

\begin{gather}
	\cos^2(q_x) + \cos(q_x)\cos(\sqrt{3}q_y) + 1/4 = 0 \\
	\Rightarrow \cos(q_x) = \frac12\left(-\cos(\sqrt{3}q_y) \pm \sqrt{\cos^2(\sqrt{3}q_y)-1}\right)
\end{gather}

A handy solution to this equation is 

\begin{align}
	\begin{split}
		q_x &= (-1)^s\frac{2\pi}{3} + m_1\pi \\
		q_y &= \frac{\pi}{\sqrt{3}}\left(2m_2-m_1\right),
	\end{split}
\end{align}

with $s \in {0,1}$ and $m_1, m_2 \in \mathbb{N}$. These touching points lie at the corners of the Brillouin zone. They are all symmetry-equivalent. Representives for the time-revearsal partners in the first Brillouin zone are 

\begin{align}
	\vec{K} &= 
	\begin{pmatrix}
		2\pi/3 \\
		0
	\end{pmatrix}, \quad s=0, \; m_1 = 0, \; m_2 = 0 \\
	\vec{K}^\prime &= 
	\begin{pmatrix}
		-2\pi/3 \\
		0
	\end{pmatrix}, \quad s = 1, \; m_1 = 0, \; m_2 = 0.
\end{align}


\paragraph{Small Momentum Expansion} \hfill \\

Linearize the Hamiltonian according to 

\begin{equation}
	H(\vec{K}+ \vec{p}) \approx \underbrace{H(\vec{K})}_{\equiv H_{\vec{K}}^0} + \underbrace{\vec{p} \cdot \left(\vec{\nabla}_{\vec{q}}H(\vec{q})\right)\bigg\rvert_{\vec{q} = \vec{K}}}_{\equiv H_{\vec{K}}^1} + \mathcal{O}(p^2).
\end{equation}

and find 

\begin{equation}
	H_{\vec{K}}^0 = 2t
	\begin{pmatrix}
		0 & \cos(K_2) & \cos(K_3) \\
		\cos(K_2) & 0 & \cos(K_1) \\
		\cos(K_3) & \cos(K_1) & 0 
	\end{pmatrix}
\end{equation}

and 


\begin{equation}
	H_{\vec{K}}^1 = -2t
	\begin{pmatrix}
		0 & \sin(K_2)p_2 & \sin(K_3)p_3 \\
		\sin(K_2)p_2 & 0 & \sin(K_1)p_1 \\
		\sin(K_3)p_3 & \sin(K_1)p_1 & 0 
	\end{pmatrix}.
\end{equation}


\paragraph{Projection} \hfill \\

The eigenvector of $H_{\vec{K}}^0$ corresponding to the flat band reads:

\begin{equation}
	\vec{v}_3(\vec{K}) = \frac{1}{\sqrt{3}}
	\begin{pmatrix}
		-1 \\
		1 \\
		1
	\end{pmatrix}.
\end{equation}

A possible (orthonormal) basis for the subspace orthogonal to $\vec{v}_3$ is 

\begin{equation}
	\vec{v}_1(\vec{K}) = \frac{1}{\sqrt{2}}
	\begin{pmatrix}
		1 \\
		1 \\ 
		0
	\end{pmatrix}, \quad
	\vec{v}_2(\vec{K}) = \frac{1}{\sqrt{6}}
	\begin{pmatrix}
		1 \\
		-1 \\
		2
	\end{pmatrix}.
\end{equation}

Projeting $H$ onto this basis yields the coefficients


\begin{align}
	\begin{split}
		M_{11}^x & = - \frac{\sqrt{3}}{2}t \\
		M_{11}^y &= - \frac{3}{2}t \\
		M_{12}^x &= - \frac32 t \\
		M_{12}^y &= \frac{\sqrt{3}}{2} t.
	\end{split}
\end{align}

\begin{equation}
	\Rightarrow v_F(\vec{K}) = \sqrt{3}t
\end{equation}

The corresponding parameters for an effective theory of the form $H = \varepsilon_0\sigma_0 +p_x \sigma_x p_y \sigma_y$ are 

\begin{align}
	\begin{split}
		\phi &= -\frac{\pi}{2} \\
		\kappa &= -\frac{\pi}{2} \\
		\theta &= \arctan(\sqrt{3}) = \frac{\pi}{3}.
	\end{split}
\end{align}

\subsubsection{Strain and Pseudo-Potentials}

Introducing strain yields the following additional expansion terms of $H$:

\begin{equation}
	\delta H_1 = -2t\beta
	\begin{pmatrix}
		0 & (u_{xx} + 2\sqrt{3}u_{yx}+3u_{yy})/8 & (u_{xx} - 2\sqrt{3}u_{yx}+3u_{yy}/8 \\
		(u_{xx} + 2\sqrt{3}u_{yx}+3u_{yy})/8 & 0 & -u_{xx}/2 \\
		(u_{xx} - 2\sqrt{3}u_{yx}+3u_{yy}/8 & - u_{xx}/2 & 0
	\end{pmatrix}
\end{equation}

\begin{equation}
	\delta H_2 = -2t
	\begin{pmatrix}
		0 & \pi(\frac{u_{xx}}{\sqrt{3}} + u_{yx})/2 & \pi(\frac{u_{xx}}{\sqrt{3}} - u_{yx} )/2 \\
		\pi(\frac{u_{xx}}{\sqrt{3}} + u_{yx})/2 & 0 & \pi \frac{u_{xx}}{\sqrt{3}} \\
		\pi(\frac{u_{xx}}{\sqrt{3}} - u_{yx} )/2 & \pi \frac{u_{xx}}{\sqrt{3}} & 0 
	\end{pmatrix}.
\end{equation}

Projecting these onto the correct basis gives 

\begin{align}
	\langle \delta H_1 \rangle_{\{\vec{w}_1, \vec{w}_2\}} &= -\frac{\beta}{2}(u_{xx} + u_{yy})\sigma_0 + \sqrt{3}\frac{\beta}{2\sqrt{3}}(u_{yy}-u_{xx}) \sigma_x + \sqrt{3} \frac{\beta}{\sqrt{3}}u_{yx} \sigma_y \\
	\langle \delta H_0 \rangle_{\{\vec{w}_1, \vec{w}_2\}} &= \sqrt{3}\frac{2\pi}{3}u_{xx}\sigma_x + \sqrt{3}\frac{2\pi}{3}u_{yx}\sigma_y.
\end{align}

Thus, the final result is 

\begin{align}
	\phi(\vec{K}) &= -\frac{\beta}{2}(u_{xx} + u_{yy}) \\
	\vec{A}(\vec{K}) &= \frac{\beta}{2\sqrt{3}}
	\begin{pmatrix}
		u_{yy}-u_{xx} \\
		2u_{yx}
	\end{pmatrix}
\end{align}

This is the result they found in \cite{Liu2020}.

\begin{equation}
	\Rightarrow \vec{B}(\vec{K}) = - \frac{\beta}{2\sqrt{3}}(\partial_x\partial_y U_x + [2\partial_x^2 - \partial_y^2] U_y)
\end{equation}
